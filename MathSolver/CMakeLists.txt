cmake_minimum_required(VERSION 3.21)
project(MathSolver VERSION 2.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(MATHSOLVER_GUI "Build the Qt desktop application" ON)
set(MATHSOLVER_BACKEND "LINKED" CACHE STRING "LINKED libqalculate, or PROCESS for Windows/MSVC")
set_property(CACHE MATHSOLVER_BACKEND PROPERTY STRINGS LINKED PROCESS)
add_library(mathsolver_core src/engine.cpp src/domain.cpp)
target_include_directories(mathsolver_core PUBLIC include)
if(MSVC)
 target_compile_options(mathsolver_core PUBLIC /utf-8 PRIVATE /W4)
 target_compile_definitions(mathsolver_core PUBLIC NOMINMAX)
else()
 target_compile_options(mathsolver_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MATHSOLVER_BACKEND STREQUAL "LINKED")
 find_path(QALCULATE_INCLUDE_DIR libqalculate/Calculator.h REQUIRED)
 find_library(QALCULATE_LIBRARY NAMES qalculate REQUIRED)
 target_sources(mathsolver_core PRIVATE src/calculus.cpp)
 target_include_directories(mathsolver_core PRIVATE ${QALCULATE_INCLUDE_DIR})
 target_compile_definitions(mathsolver_core PRIVATE MATHSOLVER_CAS_DEFINITIONS_DIR="${QALCULATE_INCLUDE_DIR}/../share/qalculate")
 target_link_libraries(mathsolver_core PUBLIC ${QALCULATE_LIBRARY})
 add_executable(mathsolver_cas_worker tools/cas_worker.cpp)
 target_link_libraries(mathsolver_cas_worker PRIVATE mathsolver_core)
else()
 if(NOT WIN32)
  message(FATAL_ERROR "PROCESS backend is intended for Windows/MSVC; use LINKED on macOS")
 endif()
 target_sources(mathsolver_core PRIVATE src/calculus_process.cpp)
endif()
include(CTest)
add_executable(audit_tests tests/audit_tests.cpp)
target_link_libraries(audit_tests PRIVATE mathsolver_core)
add_test(NAME audit_domain COMMAND audit_tests)
set_tests_properties(audit_domain PROPERTIES TIMEOUT 120)
add_executable(mathsolver_tests tests/core_tests.cpp)
target_link_libraries(mathsolver_tests PRIVATE mathsolver_core)
add_test(NAME core_regression COMMAND mathsolver_tests)
add_executable(calculus_tests tests/calculus_tests.cpp)
target_link_libraries(calculus_tests PRIVATE mathsolver_core)
foreach(category derivative integral limit parser)
 add_test(NAME calculus_${category} COMMAND calculus_tests ${category})
 set_tests_properties(calculus_${category} PROPERTIES TIMEOUT 120)
endforeach()
add_executable(textbook_tests tests/textbook_tests.cpp)
target_link_libraries(textbook_tests PRIVATE mathsolver_core)
foreach(category derivative integral limit syntax_domain)
 add_test(NAME textbook_${category} COMMAND textbook_tests ${category})
 set_tests_properties(textbook_${category} PROPERTIES TIMEOUT 120)
endforeach()
add_executable(mathsolver_experiments examples/experiments.cpp)
add_executable(extension_tests tests/extension_tests.cpp)
target_link_libraries(extension_tests PRIVATE mathsolver_core)
foreach(category integral limit polynomial)
 add_test(NAME extension_${category} COMMAND extension_tests ${category})
 set_tests_properties(extension_${category} PROPERTIES TIMEOUT 120)
endforeach()
target_link_libraries(mathsolver_experiments PRIVATE mathsolver_core)
add_executable(parameter_tests tests/parameter_tests.cpp)
target_link_libraries(parameter_tests PRIVATE mathsolver_core)
foreach(category parser integral limit guards)
 add_test(NAME parameter_${category} COMMAND parameter_tests ${category})
 set_tests_properties(parameter_${category} PROPERTIES TIMEOUT 120)
endforeach()
if(MATHSOLVER_GUI)
 find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent WebEngineWidgets)
 qt_add_executable(MathSolver WIN32 MACOSX_BUNDLE src/main.cpp src/window.cpp src/home.cpp src/plot.cpp)
 set_target_properties(MathSolver PROPERTIES OUTPUT_NAME Infinity)
 target_link_libraries(MathSolver PRIVATE mathsolver_core Qt6::Widgets Qt6::Concurrent Qt6::WebEngineWidgets)
 target_compile_definitions(MathSolver PRIVATE RESOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/resources/latex")
 qt_add_resources(MathSolver app_assets PREFIX "/" BASE resources FILES resources/icons/infinity.png)
 if(APPLE)
  target_sources(MathSolver PRIVATE resources/icons/Infinity.icns)
  set_source_files_properties(resources/icons/Infinity.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(MathSolver PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER org.mathsolver.desktop MACOSX_BUNDLE_BUNDLE_NAME Infinity MACOSX_BUNDLE_COPYRIGHT Lysandra MACOSX_BUNDLE_ICON_FILE Infinity.icns MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
  set(resource_destination "$<TARGET_BUNDLE_CONTENT_DIR:MathSolver>/Resources")
  install(TARGETS MathSolver BUNDLE DESTINATION .)
  install(DIRECTORY resources/latex DESTINATION Infinity.app/Contents/Resources)
 else()
  if(WIN32)
   target_sources(MathSolver PRIVATE resources/icons/Infinity.rc)
   target_include_directories(MathSolver PRIVATE resources/icons)
  endif()
  set(resource_destination "$<TARGET_FILE_DIR:MathSolver>/resources")
  install(TARGETS MathSolver RUNTIME DESTINATION .)
  install(DIRECTORY resources/latex DESTINATION resources)
 endif()
 add_custom_command(TARGET MathSolver POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/resources/latex "${resource_destination}/latex")
 add_custom_command(TARGET MathSolver POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/resources/qalculate "${resource_destination}/qalculate")
 if(APPLE)
  install(DIRECTORY resources/qalculate DESTINATION Infinity.app/Contents/Resources)
 else()
  install(DIRECTORY resources/qalculate DESTINATION resources)
 endif()
 if(MATHSOLVER_BACKEND STREQUAL "PROCESS")
  set(MATHSOLVER_CAS_DIRECTORY "" CACHE PATH "Folder containing mathsolver_cas_worker.exe and MinGW DLLs")
  if(MATHSOLVER_CAS_DIRECTORY)
   add_custom_command(TARGET MathSolver POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${MATHSOLVER_CAS_DIRECTORY}" "$<TARGET_FILE_DIR:MathSolver>/cas")
   install(DIRECTORY "${MATHSOLVER_CAS_DIRECTORY}/" DESTINATION cas)
  endif()
 endif()
 qt_generate_deploy_app_script(TARGET MathSolver OUTPUT_SCRIPT qt_deploy NO_UNSUPPORTED_PLATFORM_ERROR)
 if(APPLE)
  get_filename_component(mathsolver_qt_prefix "${Qt6_DIR}/../../.." ABSOLUTE)
  # conda-forge Qt uses dylibs and an unbundled WebEngine helper. Official Qt
  # framework builds are handled by macdeployqt itself.
  if(EXISTS "${mathsolver_qt_prefix}/lib/qt6/QtWebEngineProcess")
   install(PROGRAMS "${mathsolver_qt_prefix}/lib/qt6/QtWebEngineProcess" DESTINATION Infinity.app/Contents/Helpers)
   install(DIRECTORY "${mathsolver_qt_prefix}/share/qt6/resources/" DESTINATION Infinity.app/Contents/Resources/qtwebengine)
   install(DIRECTORY "${mathsolver_qt_prefix}/share/qt6/translations/qtwebengine_locales/" DESTINATION Infinity.app/Contents/Resources/qtwebengine_locales)
  endif()
 endif()
 install(SCRIPT ${qt_deploy})
 if(APPLE)
  install(SCRIPT ${CMAKE_CURRENT_SOURCE_DIR}/scripts/finish-macos.cmake)
 endif()
 install(FILES LICENSE docs/THIRD_PARTY.md DESTINATION licenses)
 install(DIRECTORY third_party/licenses/ DESTINATION licenses/dependencies)
 add_test(NAME gui_smoke COMMAND MathSolver --smoke)
 set_tests_properties(gui_smoke PROPERTIES TIMEOUT 100)
endif()
set(CPACK_PACKAGE_NAME Infinity)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
if(APPLE)
 set(CPACK_GENERATOR DragNDrop)
elseif(WIN32)
 set(CPACK_GENERATOR ZIP)
endif()
include(CPack)
